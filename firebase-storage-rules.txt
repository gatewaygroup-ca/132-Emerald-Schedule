// Paste this into Firebase console -> Storage -> Rules.
// Mirrors the database rules: anyone can view files (invoices, specs,
// documents are usually fine to be viewable by clients), but only a
// signed-in admin can upload/delete.

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /projects/{projectId}/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
